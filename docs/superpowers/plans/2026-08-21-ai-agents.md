# AI Agents Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let any signed-in user create a private AI agent by writing its name, its instructions and choosing its tools.

**Architecture:** An agent IS an Open WebUI `model` row: instructions in `params.system`, tools in `meta.toolIds`, owner in `user_id`, privacy from the absence of an `access_grant`. The page is static HTML and JavaScript calling Open WebUI's own model API with the signed-in user's session. No new table, no new backend service, no service-to-service auth.

**Tech Stack:** Vanilla JS + HTML (matching `static/cron.html` and `static/projects.html`), FastAPI (tasks service, one static route only), Postgres, Playwright for browser tests.

## Global Constraints

- Spec: `docs/superpowers/specs/2026-08-21-ai-agents-design.md`
- Deploy target: `root@46.224.193.25`, path `/root/proxy-server/`, compose `docker-compose.unified.yml`. No git on the server.
- **SSH to this box is intermittently flaky.** A completed `scp` is NOT evidence of a complete file: a transfer on 2026-08-21 silently truncated `task-panel.js` to 77,707 of 90,487 bytes. Every transfer must be followed by a `sha256sum` comparison and retried until it matches.
- After any transfer run `sed -i 's/\r$//'` on the server. This repo checks out CRLF on Windows.
- **NEVER touch, overwrite or commit `.env`.**
- No em-dashes or en-dashes in any user-facing copy.
- Commit messages: no AI attribution, no `Co-Authored-By`.
- Native tool ids, exactly: `calendar`, `documents`, `excel_creator`, `executive_dashboard`, `gdrive`, `gmail`, `remember`.
- The mcp-proxy tool server id is exactly `server:mcp-proxy`.
- `ui.default_models` is currently `null`, so the base-model default falls back to the first model the API returns.
- Agent id format: `agent-<slug>-<4 lowercase hex>`.
- Limits: name 1-60 chars, instructions 1-4000 chars, 25 agents per user (page-enforced, advisory).

```python
"""What each user actually sees in their model picker.

Run before and after flipping BYPASS_MODEL_ACCESS_CONTROL. The failure being
guarded against is silent and total: a non-admin opens the picker and it is
empty, with nothing logged anywhere.

Usage, inside the mcp-proxy container (it has PyJWT; tasks does not):
    docker exec -e PROBE_SECRET=... mcp-proxy python /tmp/model_visibility_probe.py
"""
import json
import os
import sys
import urllib.error
import urllib.request

BASE = os.environ.get("PROBE_BASE", "https://ai-ui.coolestdomain.win")
# Cloudflare 1010-blocks urllib's default User-Agent on this domain.
UA = ("Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 "
      "(KHTML, like Gecko) Chrome/126.0 Safari/537.36")


def models_for(token):
    req = urllib.request.Request(BASE + "/api/models", headers={
        "Authorization": "Bearer " + token, "User-Agent": UA})
    with urllib.request.urlopen(req, timeout=30) as r:
        body = json.load(r)
    data = body.get("data", body if isinstance(body, list) else [])
    return sorted(m.get("id") for m in data)


def main():
    out = {}
    for pair in os.environ["PROBE_TOKENS"].split(";"):
        label, token = pair.split("=", 1)
        try:
            ids = models_for(token)
            out[label] = ids
            print("%-30s %3d models" % (label, len(ids)))
        except urllib.error.HTTPError as e:
            print("%-30s HTTP %s" % (label, e.code))
            out[label] = None
    path = os.environ.get("PROBE_OUT", "/tmp/model_visibility.json")
    with open(path, "w") as fh:
        json.dump(out, fh, indent=1)
    print("written to", path)
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

- [ ] **Step 3: Write the backfill**

Create `mcp-servers/tasks/scripts/backfill_model_grants.py`:

```python
"""Give every visible model a database row and a wildcard read grant.

Prerequisite for turning model access control on. get_filtered_models treats a
model with no row as admin-only, and only 7 of 130 have one, so flipping the
bypass first would leave every non-admin with 7 models and nothing logged.

The row copies the capabilities the API reports for that model. An empty meta
would strip vision, tools and web search from models that have them, which is a
behaviour change smuggled in behind a visibility fix.

Idempotent. Safe to re-run. Run inside the mcp-proxy container (PyJWT).
"""
import json
import os
import sys
import urllib.request

BASE = os.environ.get("PROBE_BASE", "https://ai-ui.coolestdomain.win")
UA = ("Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 "
      "(KHTML, like Gecko) Chrome/126.0 Safari/537.36")


def admin_models(token):
    req = urllib.request.Request(BASE + "/api/models", headers={
        "Authorization": "Bearer " + token, "User-Agent": UA})
    with urllib.request.urlopen(req, timeout=40) as r:
        body = json.load(r)
    return body.get("data", body if isinstance(body, list) else [])


def main():
    token = os.environ["ADMIN_TOKEN"]
    rows = []
    for m in admin_models(token):
        mid = m.get("id")
        if not mid:
            continue
        rows.append({
            "id": mid,
            "name": m.get("name") or mid,
            "capabilities": (m.get("info") or {}).get("meta", {}).get("capabilities"),
        })
    # Emit SQL rather than JSON. The earlier draft wrote JSON and left a psql
    # variable for a later step to interpolate, which no step ever set.
    def q(v):
        return "'" + str(v).replace("'", "''") + "'"

    out = ["-- generated by backfill_model_grants.py; safe to re-run"]
    for r in rows:
        meta = json.dumps({"capabilities": r["capabilities"]}
                          if r["capabilities"] else {})
        out.append(
            "INSERT INTO model (id, user_id, base_model_id, name, meta, params,"
            " is_active, created_at, updated_at) SELECT " + q(r["id"]) +
            ", (SELECT id FROM \"user\" WHERE role='admin' ORDER BY created_at"
            " LIMIT 1), NULL, " + q(r["name"]) + ", " + q(meta) + "::json,"
            " '{}'::json, true, extract(epoch from now())::bigint,"
            " extract(epoch from now())::bigint WHERE NOT EXISTS (SELECT 1 FROM"
            " model m WHERE m.id = " + q(r["id"]) + ");")
    out.append(
        "INSERT INTO access_grant (id, resource_type, resource_id,"
        " principal_type, principal_id, permission, created_at)"
        " SELECT gen_random_uuid()::text, 'model', m.id, 'user', '*', 'read',"
        " extract(epoch from now())::bigint FROM model m WHERE NOT EXISTS ("
        "SELECT 1 FROM access_grant g WHERE g.resource_type='model'"
        " AND g.resource_id=m.id AND g.principal_type='user'"
        " AND g.principal_id='*' AND g.permission='read');")

    path = os.environ.get("BACKFILL_OUT", "/tmp/apply_model_rows.sql")
    with open(path, "w") as fh:
        fh.write("\n".join(out) + "\n")
    print("described %d models -> %s" % (len(rows), path))
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```js
    var NATIVE_TOOLS = [
      ["gmail", "Gmail"], ["calendar", "Calendar"], ["gdrive", "Drive"],
      ["documents", "Documents"], ["excel_creator", "Excel"],
      ["executive_dashboard", "Dashboard"], ["remember", "Memory"]
    ];
    var MAX_INSTRUCTIONS = 4000;
    var MAX_AGENTS = 25;
    var editingId = null;

    function slug(name) {
      return String(name).toLowerCase().replace(/[^a-z0-9]+/g, "-")
        .replace(/^-+|-+$/g, "").slice(0, 40) || "agent";
    }

    function newAgentId(name) {
      var hex = "0123456789abcdef", suffix = "";
      for (var i = 0; i < 4; i++) suffix += hex[Math.floor(Math.random() * 16)];
      return "agent-" + slug(name) + "-" + suffix;
    }

    function chosenTools() {
      var ids = [];
      if (document.getElementById("use-my-apps").checked) ids.push("server:mcp-proxy");
      NATIVE_TOOLS.forEach(function (t) {
        if (document.getElementById("tool-" + t[0]).checked) ids.push(t[0]);
      });
      return ids;
    }

    // The exact ModelForm the API expects. Instructions live in params.system
    // and tools in meta.toolIds; both are extra fields on models that allow
    // extras, which is how Open WebUI's own builder stores them.
    function buildAgentBody(f) {
      return {
        id: f.id, name: f.name, base_model_id: f.baseModel,
        meta: { description: f.instructions.slice(0, 120), toolIds: f.toolIds },
        params: { system: f.instructions },
        is_active: true
      };
    }

    function formError(msg) { document.getElementById("form-error").textContent = msg; }

    function openForm(agent) {
      editingId = agent ? agent.id : null;
      document.getElementById("form-title").textContent =
        agent ? "Edit agent" : "New agent";
      document.getElementById("agent-name").value = agent ? agent.name : "";
      document.getElementById("agent-instructions").value =
        agent && agent.params ? (agent.params.system || "") : "";
      var tools = (agent && agent.meta && agent.meta.toolIds) || [];
      document.getElementById("use-my-apps").checked =
        tools.indexOf("server:mcp-proxy") !== -1;
      NATIVE_TOOLS.forEach(function (t) {
        document.getElementById("tool-" + t[0]).checked = tools.indexOf(t[0]) !== -1;
      });
      var base = document.getElementById("agent-base");
      base.innerHTML = "";
      state.models.filter(function (m) { return !isAgent(m); }).forEach(function (m) {
        var o = document.createElement("option");
        o.value = m.id; o.textContent = m.name || m.id;
        base.appendChild(o);
      });
      if (agent && agent.base_model_id) base.value = agent.base_model_id;
      formError("");
      updateChars();
      document.getElementById("agent-overlay").hidden = false;
    }

    function closeForm() { document.getElementById("agent-overlay").hidden = true; }

    function updateChars() {
      document.getElementById("chars").textContent =
        String(document.getElementById("agent-instructions").value.length);
    }

    async function save() {
      var name = document.getElementById("agent-name").value.trim();
      var instructions = document.getElementById("agent-instructions").value.trim();
      if (!name) return formError("Give the agent a name.");
      if (!instructions) return formError("Tell the agent what to do.");
      if (instructions.length > MAX_INSTRUCTIONS) {
        return formError("Instructions are " + instructions.length +
                         " characters. The limit is 4000.");
      }
      var mine = state.agents.filter(function (m) { return m.user_id === state.me; });
      if (!editingId && mine.length >= MAX_AGENTS) {
        return formError("You have " + MAX_AGENTS + " agents, which is the limit. " +
                         "Delete one to make another.");
      }
      var body = buildAgentBody({
        id: editingId || newAgentId(name), name: name,
        baseModel: document.getElementById("agent-base").value,
        instructions: instructions, toolIds: chosenTools()
      });
      var url = API + (editingId ? "/model/update?id=" + encodeURIComponent(editingId)
                                 : "/create");
      var btn = document.getElementById("agent-save");
      btn.disabled = true;
      try {
        var r = await fetch(url, { method: "POST", headers: authHeaders(),
                                   body: JSON.stringify(body) });
        // The id is a primary key across every model on the platform, and four
        // hex characters can collide. One retry with a fresh suffix, and only
        // when creating: on edit the id is deliberately the existing one.
        if (!editingId && (r.status === 400 || r.status === 409)) {
          body.id = newAgentId(name);
          r = await fetch(API + "/create", { method: "POST",
                headers: authHeaders(), body: JSON.stringify(body) });
        }
        if (r.status === 401) throw new Error("Agents are not switched on for " +
                                              "your account yet.");
        if (!r.ok) throw new Error("Could not save that agent.");
        closeForm();
        await load();
      } catch (e) {
        // Never clear the form on failure: the user typed that.
        formError(e.message || "Could not save that agent.");
      } finally {
        btn.disabled = false;
      }
    }
```

```python
"""Create the ready-made agents everyone gets.

A platform agent is an ordinary agent with a wildcard read grant, the same
mechanism the base models use. Everyone sees it; only its owner can edit it.

Idempotent: an agent that already exists is updated rather than duplicated.
Run inside mcp-proxy with ADMIN_TOKEN set.
"""
import json
import os
import sys
import urllib.error
import urllib.request

BASE = "https://ai-ui.coolestdomain.win"
UA = "Mozilla/5.0"

AGENTS = [
    {
        "id": "agent-research-assistant-0001",
        "name": "Research Assistant",
        "system": (
            "You research questions carefully and answer with what you found, "
            "not with what you assume. Search the web when the answer depends "
            "on current facts. Say plainly when you could not find something, "
            "and never present a guess as a finding. Keep answers short and "
            "put the conclusion first."
        ),
        "tools": ["server:mcp-proxy"],
    },
    {
        "id": "agent-inbox-triage-0002",
        "name": "Inbox Triage",
        "system": (
            "You read the user's unread email and tell them what actually "
            "needs them. Group messages into: needs a reply today, can wait, "
            "and no action. Give one line per message with who it is from and "
            "what they want. Never send or delete anything without being "
            "asked to."
        ),
        "tools": ["gmail"],
    },
]


def post(path, body, token):
    req = urllib.request.Request(
        BASE + path, data=json.dumps(body).encode(),
        headers={"Content-Type": "application/json", "User-Agent": UA,
                 "Authorization": "Bearer " + token})
    return urllib.request.urlopen(req, timeout=30)


def main():
    token = os.environ["ADMIN_TOKEN"]
    for a in AGENTS:
        body = {
            "id": a["id"], "name": a["name"], "base_model_id": None,
            "meta": {"description": a["system"][:120], "toolIds": a["tools"]},
            "params": {"system": a["system"]},
            "access_grants": [{"principal_type": "user", "principal_id": "*",
                               "permission": "read"}],
            "is_active": True,
        }
        try:
            post("/api/v1/models/create", body, token)
            print("created", a["id"])
        except urllib.error.HTTPError as e:
            if e.code in (400, 409):
                post("/api/v1/models/model/update?id=" + a["id"], body, token)
                print("updated", a["id"])
            else:
                print("FAILED", a["id"], e.code)
                return 1
    return 0


if __name__ == "__main__":
    sys.exit(main())
```
